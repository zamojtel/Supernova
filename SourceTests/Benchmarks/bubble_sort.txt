void bubble_sort(int * arr,int n) {
	for (int i = 0; i < n ;i++) {
		bool swapped = false;
		int new_limit = n - i;
		for (int j = 1; j < new_limit ;j++) {
			if (arr[j-1]>arr[j]) {
				int temp =  arr[j-1];
				arr[j-1] = arr[j];
				arr[j] = temp;
				swapped = true;
			}
		}
		if (swapped==false)
			return;
	}
	return;
}
