void merge_sort_impl(int* a, int* temp, int left, int right) {
	if (right - left <= 1)
		return;
	int const mid = left + (right - left) / 2;
	merge_sort_impl(a, temp, left, mid);
	merge_sort_impl(a, temp, mid, right);

	int i = left, j = mid, k = left;
	while (i < mid && j < right) {
		if (a[i] <= a[j])
			temp[k++] = a[i++];
		else
			temp[k++] = a[j++];
	}

	while (i < mid)
		temp[k++] = a[i++];

	while (j < right)
		temp[k++] = a[j++];

	for (int p = left; p < right; ++p)
		a[p] = temp[p];
}

void merge_sort(int* a, int n) {
	int* temp = (int*)malloc(((uint64)n) * sizeof(int));
	merge_sort_impl(a, temp, 0, n);
	free(temp);
}
