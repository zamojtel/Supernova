float factorial(int n){
    float result = 1;
    if(n<=1)
        return 1;

    for(int i = 2;i <= n; i++){
        result = result  * i;
    }

    return result;
}

int main() {
    float [4] data;
    data[0] = -5;
    data[1] = 10;
    data[2] = 0;
    data[3] = 7;
    bool [4] result = data >= 0.0f;
    float [4] sign = select(result, factorial(3), factorial(4));
    print(sign);
    return 0;
}
